# ================================================
# Legacy temporary trycloudflare.com tunnel (fallback only)
# ================================================
# Use this only when deliberately running a temporary Quick Tunnel. It has a
# changing URL and requires the Vercel update/deploy steps below.
#
# Important: use chatbot-api-quick-tunnel for this legacy tunnel. Do NOT use
# chatbot-api-tunnel, which is the permanent Cloudflare Named Tunnel above.
cd /home/huor/Documents/gdce-document/chatbot_review

# 1. Check the temporary tunnel.
TUNNEL_URL=$(docker logs chatbot-api-quick-tunnel 2>&1 | grep -oE 'https://[^[:space:]]+\.trycloudflare\.com' | tail -1)
echo "$TUNNEL_URL"
curl -i "$TUNNEL_URL/api/v1/health"
# HTTP 200 means it works. Stop here.

# 2. If it fails, restart the temporary tunnel.
docker restart chatbot-api-quick-tunnel

# 3. Check the new temporary URL.
TUNNEL_URL=$(docker logs chatbot-api-quick-tunnel 2>&1 | grep -oE 'https://[^[:space:]]+\.trycloudflare\.com' | tail -1)
echo "$TUNNEL_URL"
curl -i "$TUNNEL_URL/api/v1/health"
# Continue only with HTTP 200.

# 4. Update Vercel to the new temporary URL, then deploy.
vercel whoami || vercel login
vercel link --yes --project customs-artificial-intelligence --scope vuth-menghuors-projects
printf '%s\n' "$TUNNEL_URL" | vercel env update NUXT_PUBLIC_GDCE_API_BASE_URL production --yes
printf '%s\n' "$TUNNEL_URL" | vercel env update NUXT_PUBLIC_GDCE_API_BASE_URL preview --yes
vercel --prod --yes

# 5. Verify the deployed login page.
curl -s https://customs-artificial-intelligence.vercel.app/login | grep -o 'gdceApiBaseUrl:"[^"]*"'
